**2026-10-01, Lead PO — the draw scripts, verbatim.** Run from a scratch
directory outside the repository. `dims.sh` reads dimensions from the first
64 KiB of a PNG, JPEG or WebP without decoding. `draw.awk` takes the corpus
stems, then the pool keyed and sorted by `(folder, sha256 key)`, and is run
as `awk -F'\t' -v TARGET=15 -f draw.awk corpus_plus_fix.txt keyed.tsv` (GNU Awk
5.4.1, for `mktime` and `match` with an array).

`dims.sh`:

```bash
#!/usr/bin/env bash
# Print "WxH" for a PNG, JPEG or WebP from its header bytes only (no decode).
f="$1"
hex=$(head -c 65536 "$f" | od -An -v -tx1 | tr -s ' \n' ' ')
set -- $hex
b() { echo $((16#${@})); }
if [ "$1$2$3$4" = "89504e47" ]; then
  echo "$(( (16#${17}<<24)|(16#${18}<<16)|(16#${19}<<8)|16#${20} ))x$(( (16#${21}<<24)|(16#${22}<<16)|(16#${23}<<8)|16#${24} ))"; exit
fi
if [ "$1$2" = "ffd8" ]; then
  arr=($hex); i=2; n=${#arr[@]}
  while [ $i -lt $((n-9)) ]; do
    [ "${arr[$i]}" != "ff" ] && { i=$((i+1)); continue; }
    m=${arr[$((i+1))]}
    case $m in c0|c1|c2|c3|c5|c6|c7|c9|ca|cb|cd|ce|cf)
      h=$(( (16#${arr[$((i+5))]}<<8)|16#${arr[$((i+6))]} )); w=$(( (16#${arr[$((i+7))]}<<8)|16#${arr[$((i+8))]} )); echo "${w}x${h}"; exit;;
    d8|01|d0|d1|d2|d3|d4|d5|d6|d7) i=$((i+2)); continue;;
    esac
    len=$(( (16#${arr[$((i+2))]}<<8)|16#${arr[$((i+3))]} )); i=$((i+2+len))
  done
  echo "?"; exit
fi
if [ "$1$2$3$4" = "52494646" ] && [ "$9${10}${11}${12}" = "57454250" ]; then
  arr=($hex); chunk="${arr[12]}${arr[13]}${arr[14]}${arr[15]}"
  case $chunk in
    56503858) w=$(( 1 + (16#${arr[24]} | 16#${arr[25]}<<8 | 16#${arr[26]}<<16) )); h=$(( 1 + (16#${arr[27]} | 16#${arr[28]}<<8 | 16#${arr[29]}<<16) ));;
    56503820) w=$(( (16#${arr[26]} | 16#${arr[27]}<<8) & 0x3fff )); h=$(( (16#${arr[28]} | 16#${arr[29]}<<8) & 0x3fff ));;
    5650384c) b0=$((16#${arr[21]})); b1=$((16#${arr[22]})); b2=$((16#${arr[23]})); b3=$((16#${arr[24]})); w=$(( 1 + (b0 | (b1 & 0x3f)<<8) )); h=$(( 1 + ((b1>>6) | b2<<2 | (b3 & 0x0f)<<10) ));;
    *) echo "?"; exit;;
  esac
  echo "${w}x${h}"; exit
fi
echo "?"
```

`draw.awk`:

```awk
# Round-robin draw with the MC-062 no-twin rule. Pass 1: corpus stems; pass 2: keyed pool.
function tmin(s,   a) { if (match(s, /^([0-9]{4})-([0-9]{2})-([0-9]{2}) ([0-9]{2})_([0-9]{2})_([0-9]{2})/, a)) return mktime(a[1]" "a[2]" "a[3]" "a[4]" "a[5]" "a[6]) / 60; return "" }
function snum(s,   a) { if (match(s, /^Screenshot \(([0-9]+)\)/, a)) return a[1] + 0; return "" }
function twin(s,   t, n, i) {
  t = tmin(s); n = snum(s)
  if (t != "") for (i = 1; i <= nt; i++) if (t - T[i] <= 30 && T[i] - t <= 30) return 1
  if (n != "") for (i = 1; i <= nn; i++) if (n - N[i] <= 20 && N[i] - n <= 20) return 1
  return 0 }
function add(s,   t, n) { t = tmin(s); n = snum(s); if (t != "") T[++nt] = t; if (n != "") N[++nn] = n }
FNR == NR { add($0); next }
{ split($0, f, "\t"); fo = f[1]; cnt[fo]++; L[fo, cnt[fo]] = $0 }
END {
  order[1] = "mahwa panels"; order[2] = "Eleceed"; order[3] = "Hero Killer"; order[4] = "Unholy Blood"
  for (j = 1; j <= 4; j++) pos[order[j]] = 1
  drawn = 0; skipped = 0
  while (drawn < TARGET) {
    progress = 0
    for (j = 1; j <= 4 && drawn < TARGET; j++) {
      fo = order[j]
      while (pos[fo] <= cnt[fo]) {
        split(L[fo, pos[fo]], f, "\t"); pos[fo]++
        if (twin(f[3])) { skipped++; continue }
        add(f[3]); drawn++; took[fo]++; progress = 1
        printf "%d\t%s\t%s\t%s\t%s\n", drawn, f[3], fo, f[4], f[5]; break
      }
    }
    if (!progress) break
  }
  for (j = 1; j <= 4; j++) printf "# %s: %d drawn\n", order[j], took[order[j]] > "/dev/stderr"
  printf "# skipped as near-twins: %d\n", skipped > "/dev/stderr"
}
```
